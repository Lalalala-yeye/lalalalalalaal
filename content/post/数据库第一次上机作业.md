# Q1
```SQL
CREATE TABLE dp (
    dp_id      VARCHAR(20) not null PRIMARY KEY,      -- 部门编号
    dp_name    VARCHAR(100) NOT NULL,        -- 部门名
    floor      INT        -- 楼层
);

-- 2) 员工表
CREATE TABLE employee (
    em_id       VARCHAR(20)  not null PRIMARY KEY,      -- 员工编号
    em_name     VARCHAR(50) NOT NULL,         -- 员工姓名
    em_salary   NUMERIC(10,2) NOT null,          -- 员工薪资
    absence_days	INT not null DEFAULT 0, -- 员工缺勤天数
    birth_day	DATE, -- 员工生日
    if_married	BOOL, -- 员工婚姻情况
    dp_id	VARCHAR(20) not null, 	-- 员工所属部门
    CONSTRAINT chk_em_salary check (em_salary >= 2000 )
);

-- 加外键
ALTER TABLE employee
  ADD CONSTRAINT fk_dp_employee
  FOREIGN KEY (dp_id) REFERENCES dp(dp_id);
```

# Q3-Q13
```SQL
-- Q3
SELECT em_salary, absence_days
FROM employee;

-- Q4
SELECT COUNT(*) AS employee_count
FROM employee;

-- Q5
SELECT AVG(em_salary) AS avg_salary
FROM employee;

-- Q6
SELECT dp_id,
       MAX(em_salary) AS max_salary,
       MIN(em_salary) AS min_salary
FROM employee
WHERE dp_id = 'DP007'
GROUP BY dp_id;

-- Q7
SELECT em_name
FROM employee
WHERE absence_days > 3;

-- Q9
SELECT UPPER(em_name) AS em_name_upper
FROM employee;

-- Q10（openGauss）
SELECT em_id,
       em_name,
       birth_day,
       TO_CHAR(birth_day, 'YYYY/MM/DD')   AS birth_fmt_slash,
       TO_CHAR(birth_day, 'YYYYMMDD')     AS birth_fmt_compact
FROM employee;

-- Q11（openGauss）选定员工：412502
SELECT em_name,
       SUBSTR(em_name, 1, POSITION(' ' IN em_name) - 1) AS part1,
       SUBSTR(em_name, POSITION(' ' IN em_name) + 1)    AS part2
FROM employee
WHERE em_id = '412502';

SELECT AGE(e1.birth_day, e2.birth_day) AS age_interval
FROM employee e1, employee e2
WHERE e1.em_id = '112233' AND e2.em_id = '904862';

-- Q13（openGauss）
SELECT em_id, em_name, birth_day
FROM employee
WHERE birth_day > TO_DATE('2000-01-01', 'YYYY-MM-DD');
```
